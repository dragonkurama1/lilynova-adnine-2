// ── Construction d'une ligne produit ─────────────────────────
function makeRow(id, nom, collection, couleur, sizesActives, date, image, prix, prixPromo) {
  // sizesActives = tableau des tailles applicables, ex: ['S','M','L','XL','XXL']
  return [
    id, nom, collection, couleur, image || '',
    ...ALL_SIZES.map(s => sizesActives.includes(s) ? 0 : ''),
    date,
    Utilities.getUuid(),       // VariantId — identifiant stable de la variante
    prix != null ? Number(prix) : '',
    prixPromo != null ? Number(prixPromo) : '',
    'actif'                    // Statut par défaut
  ]
}

// ── Créer l'onglet Stock ──────────────────────────────────────
function setupStockSheet(ss) {
  const existing = ss.getSheetByName(STOCK_SHEET)
  if (existing) ss.deleteSheet(existing)

  const sheet = ss.insertSheet(STOCK_SHEET, 0)
  sheet.setTabColor('#1D4ED8')

  // En-tête
  sheet.appendRow(STOCK_HEADERS)
  const hRange = sheet.getRange(1, 1, 1, STOCK_HEADERS.length)
  hRange.setFontFamily('Arial').setFontSize(10).setFontWeight('bold')
        .setFontColor('#FFFFFF').setBackground('#1A1A1A')
        .setHorizontalAlignment('center').setVerticalAlignment('middle')

  // Tailles vêtements → bleu
  sheet.getRange(1, 6, 1, 6).setBackground('#1D4ED8')
  // Tailles lingerie → violet
  sheet.getRange(1, 12, 1, 6).setBackground('#7C3AED')

  sheet.setFrozenRows(1)
  sheet.setRowHeight(1, 32)
  hRange.setBorder(true,true,true,true,true,true,'#374151', SpreadsheetApp.BorderStyle.SOLID)

  // Largeurs
  sheet.setColumnWidth(1, 80)
  sheet.setColumnWidth(2, 230)
  sheet.setColumnWidth(3, 160)
  sheet.setColumnWidth(4, 150)
  sheet.setColumnWidth(5, 220)
  for (let c = 6; c <= 17; c++) sheet.setColumnWidth(c, 52)
  sheet.setColumnWidth(18, 110)
  sheet.setColumnWidth(19, 220) // VariantId
  sheet.setColumnWidth(20, 80)  // Prix
  sheet.setColumnWidth(21, 80)  // PrixPromo
  sheet.setColumnWidth(22, 90)  // Statut

  // Charger tous les produits
  const rows = getInitialProducts()
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, STOCK_HEADERS.length).setValues(rows)
  }

  // Formatage des lignes produits
  for (let r = 2; r <= rows.length + 1; r++) {
    sheet.setRowHeight(r, 22)
    const bg = (r % 2 === 0) ? '#FFFFFF' : '#F9FAFB'
    const dataRange = sheet.getRange(r, 1, 1, STOCK_HEADERS.length)
    dataRange.setBackground(bg).setFontFamily('Arial').setFontSize(10)
             .setVerticalAlignment('middle')
             .setBorder(true,true,true,true,true,true,'#E5E7EB', SpreadsheetApp.BorderStyle.SOLID)

    // ID en gras
    sheet.getRange(r, 1).setFontWeight('bold')
    // Tailles centrées
    sheet.getRange(r, 6, 1, 13).setHorizontalAlignment('center').setFontColor('#374151')
    // Mise à jour centrée
    sheet.getRange(r, 18).setHorizontalAlignment('center').setFontColor('#6B7280').setFontSize(9)
  }

  // Validation numérique sur colonnes tailles (empêcher texte)
  const numRule = SpreadsheetApp.newDataValidation()
    .requireNumberGreaterThanOrEqualTo(0).setAllowInvalid(false).build()
  sheet.getRange(2, 6, 500, 13).setDataValidation(numRule)

  // Légende en bas
  const legendRow = rows.length + 3
  sheet.getRange(legendRow, 1).setValue('Légende stock :').setFontWeight('bold').setFontSize(9).setFontColor('#374151')
  const legendData = [
    [legendRow, 2, '🟢 > 5 unités', '#10B981'],
    [legendRow, 3, '🟡 1–5 unités', '#F59E0B'],
    [legendRow, 4, '🔴 0 = Rupture', '#EF4444'],
    [legendRow, 5, '— = Non applicable (laisser vide)', '#9CA3AF'],
  ]
  legendData.forEach(([row, col, text, color]) => {
    const cell = sheet.getRange(row, col)
    cell.setValue(text).setFontColor(color).setFontWeight('bold').setFontSize(9)
  })

  // Note sur les collections
  const noteRow = legendRow + 2
  sheet.getRange(noteRow, 1).setValue('⚠️ Note :').setFontWeight('bold').setFontSize(9).setFontColor('#D97706')
  sheet.getRange(noteRow, 2, 1, 6).merge()
       .setValue('Les stocks sont initialisés à 0. Mets tes vrais stocks dans les colonnes correspondantes, puis les clients voient en temps réel.')
       .setFontSize(9).setFontColor('#92400E').setFontStyle('italic')
       .setBackground('#FFFBEB')
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents)
    switch (data.action) {
      case 'updateStock':      return updateStock(data)
      case 'addProduct':       return addProduct(data)
      case 'updateProduct':    return updateProduct(data)
      case 'deleteProduct':    return deleteProduct(data)
      case 'uploadImage':      return uploadImage(data)
      case 'addCatalogue':     return addCatalogue(data)
      case 'updateCatalogue':  return updateCatalogue(data)
      case 'deleteCatalogue':  return deleteCatalogue(data)
      default:                 return saveOrder(data)
    }
  } catch (err) {
    return json({ result: 'error', message: err.toString() })
  }
}

function doGet(e) {
  try {
    const action = e.parameter.action
    switch (action) {
      case 'getStock':        return getStock()
      case 'getProducts':     return getProducts()
      case 'getCollections':  return getCollections()
      case 'getCatalogues':   return getCatalogues()
      default:                return json({ result: 'error', message: 'Action inconnue: ' + action })
    }
  } catch (err) {
    return json({ result: 'error', message: err.toString() })
  }
}

function ensureStockSheet() {
  const ss    = SpreadsheetApp.getActiveSpreadsheet()
  let sheet   = ss.getSheetByName(STOCK_SHEET)
  if (!sheet) { setupStockSheet(ss); sheet = ss.getSheetByName(STOCK_SHEET) }
  migrateStockSheet(sheet)
  return sheet
}

// ── Migration / compatibilité avec les anciennes lignes ────────
// Répare automatiquement les feuilles créées avant v6 :
//  - ajoute les colonnes manquantes (VariantId, Prix, PrixPromo, Statut)
//    sans toucher aux colonnes existantes (aucun décalage d'index)
//  - génère un VariantId pour toute ligne qui n'en a pas encore
//  - met Statut = 'actif' pour toute ligne qui n'a pas de statut
//  - ne crée jamais de doublon, ne supprime jamais de données
function migrateStockSheet(sheet) {
  const lastCol = sheet.getLastColumn()
  if (lastCol === 0) return
  const headerRange = sheet.getRange(1, 1, 1, lastCol)
  let headers = headerRange.getValues()[0]

  // 1) Ajouter les colonnes manquantes à la fin, sans rien décaler
  const missing = STOCK_HEADERS.filter(h => headers.indexOf(h) === -1)
  if (missing.length > 0) {
    Logger.log('[migrateStockSheet] colonnes manquantes détectées, ajout : ' + missing.join(', '))
    const startCol = sheet.getLastColumn() + 1
    sheet.getRange(1, startCol, 1, missing.length).setValues([missing])
    headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]
  }

  const lastRow = sheet.getLastRow()
  if (lastRow < 2) return

  const variantCol = headers.indexOf('VariantId') + 1
  const statutCol  = headers.indexOf('Statut') + 1
  const idCol      = headers.indexOf('ID') + 1

  const variantVals = sheet.getRange(2, variantCol, lastRow - 1, 1).getValues()
  const statutVals  = sheet.getRange(2, statutCol, lastRow - 1, 1).getValues()
  const idVals      = sheet.getRange(2, idCol, lastRow - 1, 1).getValues()

  let backfilled = 0
  for (let i = 0; i < idVals.length; i++) {
    if (!idVals[i][0]) continue // ligne vide, on l'ignore
    if (!variantVals[i][0]) {
      variantVals[i][0] = Utilities.getUuid()
      backfilled++
    }
    if (!statutVals[i][0]) {
      statutVals[i][0] = 'actif'
    }
  }

  if (backfilled > 0) {
    Logger.log('[migrateStockSheet] VariantId généré pour ' + backfilled + ' ancienne(s) ligne(s).')
  }
  sheet.getRange(2, variantCol, lastRow - 1, 1).setValues(variantVals)
  sheet.getRange(2, statutCol, lastRow - 1, 1).setValues(statutVals)
}

// ── Lire tous les produits ────────────────────────────────────
// ── Cache (perf) ─────────────────────────────────────────────
// Évite de relire/reconstruire toute la feuille à chaque appel.
// Invalidé automatiquement par chaque écriture (add/update/delete).
const CACHE_TTL_SEC = 50
function clearProductsCache()   { try { CacheService.getScriptCache().remove('products_v1') } catch (e) {} }
function clearCataloguesCache() { try { CacheService.getScriptCache().remove('catalogues_v1') } catch (e) {} }

function getProducts() {
  const cache  = CacheService.getScriptCache()
  const cached = cache.get('products_v1')
  if (cached) return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON)

  const sheet   = ensureStockSheet()
  const allData = sheet.getDataRange().getValues()
  if (allData.length <= 1) return json({ success: true, products: [] })
  const headers = allData[0]
  const rows    = []
  let skippedDeleted = 0
  let skippedEmpty   = 0
  for (let i = 1; i < allData.length; i++) {
    const row = {}
    headers.forEach((h, j) => { row[h] = allData[i][j] })
    if (!row['ID']) { skippedEmpty++; continue }
    // Filtre défensif : ne jamais renvoyer une ligne marquée supprimée
    if (String(row['Statut'] || '').trim().toLowerCase() === 'supprime' ||
        String(row['Statut'] || '').trim().toLowerCase() === 'supprimé') {
      skippedDeleted++
      continue
    }
    rows.push(row)
  }
  Logger.log('[getProducts] lignes renvoyées=' + rows.length + ' | ignorées (vides)=' + skippedEmpty + ' | ignorées (supprimées)=' + skippedDeleted)
  const result = JSON.stringify({ success: true, products: rows })
  try { cache.put('products_v1', result, CACHE_TTL_SEC) } catch (e) {}
  return ContentService.createTextOutput(result).setMimeType(ContentService.MimeType.JSON)
}

// ── Lire tous les catalogues (triés par Ordre puis Nom) ────────
function getCatalogues() {
  const cache  = CacheService.getScriptCache()
  const cached = cache.get('catalogues_v1')
  if (cached) return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON)

  const sheet   = ensureCataloguesSheet()
  const allData = sheet.getDataRange().getValues()
  if (allData.length <= 1) return json({ success: true, catalogues: [] })
  const headers = allData[0]
  const rows    = []
  for (let i = 1; i < allData.length; i++) {
    const row = {}
    headers.forEach((h, j) => { row[h] = allData[i][j] })
    if (!row['Nom']) continue
    rows.push(row)
  }
  rows.sort((a, b) => {
    const oa = a['Ordre'] === '' || a['Ordre'] == null ? 999 : Number(a['Ordre'])
    const ob = b['Ordre'] === '' || b['Ordre'] == null ? 999 : Number(b['Ordre'])
    if (oa !== ob) return oa - ob
    return String(a['Nom']).localeCompare(String(b['Nom']))
  })
  const result = JSON.stringify({ success: true, catalogues: rows })
  try { cache.put('catalogues_v1', result, CACHE_TTL_SEC) } catch (e) {}
  return ContentService.createTextOutput(result).setMimeType(ContentService.MimeType.JSON)
}

// ── Ajouter un catalogue ────────────────────────────────────────
function addCatalogue(data) {
  const sheet   = ensureCataloguesSheet()
  const allData = sheet.getDataRange().getValues()
  const headers = allData[0]
  const nomCol  = headers.indexOf('Nom')

  if (!data.Nom || String(data.Nom).trim() === '') {
    return json({ success: false, error: 'Le champ "Nom" est obligatoire.' })
  }

  for (let i = 1; i < allData.length; i++) {
    if (norm(allData[i][nomCol]) === norm(data.Nom)) {
      return json({ success: false, error: 'Un catalogue avec ce nom existe déjà.' })
    }
  }

  const newRow = new Array(headers.length).fill('')
  headers.forEach((h, j) => {
    if (h === 'ID') {
      newRow[j] = data.ID || Utilities.getUuid()
    } else if (h === 'Ordre') {
      newRow[j] = (data.Ordre != null && data.Ordre !== '') ? Number(data.Ordre) : ''
    } else if (data[h] != null) {
      newRow[j] = data[h]
    }
  })

  sheet.appendRow(newRow)
  Logger.log('[addCatalogue] catalogue créé : ' + data.Nom)
  clearCataloguesCache()
  return json({ success: true, created: true })
}

// ── Modifier un catalogue (par ID, fallback par Nom) ────────────
function updateCatalogue(data) {
  const sheet   = ensureCataloguesSheet()
  const allData = sheet.getDataRange().getValues()
  const headers = allData[0]
  const idCol   = headers.indexOf('ID')
  const nomCol  = headers.indexOf('Nom')

  let targetRow = -1
  for (let i = 1; i < allData.length; i++) {
    if (data.ID && String(allData[i][idCol]).trim() === String(data.ID).trim()) {
      targetRow = i + 1
      break
    }
  }
  if (targetRow === -1 && data.Nom) {
    for (let i = 1; i < allData.length; i++) {
      if (norm(allData[i][nomCol]) === norm(data.Nom)) {
        targetRow = i + 1
        break
      }
    }
  }

  if (targetRow === -1) {
    return json({ success: false, error: 'Catalogue introuvable (ID ou Nom).' })
  }

  headers.forEach((h, j) => {
    if (h === 'ID') return
    if (h === 'Ordre') {
      if (data.Ordre != null && data.Ordre !== '') {
        sheet.getRange(targetRow, j + 1).setValue(Number(data.Ordre))
      }
    } else if (data[h] != null) {
      sheet.getRange(targetRow, j + 1).setValue(data[h])
    }
  })

  Logger.log('[updateCatalogue] ligne ' + targetRow + ' mise à jour')
  clearCataloguesCache()
  return json({ success: true, updated: true })
}

// ── Normalisation : trim + lowercase + espaces multiples écrasés ─
function norm(v) {
  return String(v == null ? '' : v).trim().toLowerCase().replace(/\s+/g, ' ')
}

// ── Ajouter un produit ────────────────────────────────────────
function addProduct(data) {
  const sheet   = ensureStockSheet()
  const allData = sheet.getDataRange().getValues()
  const headers = allData[0]
  const idCol    = headers.indexOf('ID')
  const colorCol = headers.indexOf('Couleur')

  Logger.log('[addProduct] données reçues = ' + JSON.stringify(data))

  // Vérifier doublon ID + Couleur (comparaison normalisée : trim + lowercase)
  for (let i = 1; i < allData.length; i++) {
    if (norm(allData[i][idCol]) === norm(data.ID) &&
        norm(allData[i][colorCol]) === norm(data.Couleur)) {
      Logger.log('[addProduct] doublon détecté pour ID=' + data.ID + ' Couleur=' + data.Couleur)
      return json({ success: false, error: 'Cette combinaison ID + Couleur existe déjà.' })
    }
  }

  const newRow = new Array(headers.length).fill('')
  headers.forEach((h, j) => {
    if (h === 'Mise à jour') {
      newRow[j] = Utilities.formatDate(new Date(), 'Africa/Casablanca', 'yyyy-MM-dd')
    } else if (h === 'VariantId') {
      newRow[j] = data.VariantId || Utilities.getUuid()
    } else if (h === 'Statut') {
      newRow[j] = data.Statut || 'actif'
    } else if (h === 'Prix' || h === 'PrixPromo') {
      newRow[j] = (data[h] != null && data[h] !== '') ? Number(data[h]) : ''
    } else if (ALL_SIZES.includes(h)) {
      newRow[j] = (data[h] != null && data[h] !== '') ? Number(data[h]) : ''
    } else if (data[h] != null) {
      newRow[j] = data[h]
    }
  })

  sheet.appendRow(newRow)

  // Formatage rapide de la nouvelle ligne
  const r = sheet.getLastRow()
  sheet.setRowHeight(r, 22)
  const bg = (r % 2 === 0) ? '#FFFFFF' : '#F9FAFB'
  sheet.getRange(r, 1, 1, headers.length)
       .setBackground(bg).setFontFamily('Arial').setFontSize(10)
       .setVerticalAlignment('middle')
       .setBorder(true,true,true,true,true,true,'#E5E7EB', SpreadsheetApp.BorderStyle.SOLID)
  sheet.getRange(r, 1).setFontWeight('bold')
  sheet.getRange(r, 6, 1, 13).setHorizontalAlignment('center')

  Logger.log('[addProduct] ligne créée à la position ' + r + ' avec VariantId=' + newRow[headers.indexOf('VariantId')])
  clearProductsCache()
  return json({ success: true, created: true })
}

// ── Recherche robuste d'une variante ────────────────────────────
// Stratégie : 1) VariantId exact  2) ID + Couleur normalisés (trim+lowercase)
// Ne renvoie jamais d'exception : toujours { rowIndex, debug } ou rowIndex=-1
function findVariantRow(sheet, data) {
  const allData = sheet.getDataRange().getValues()
  const headers = allData[0]
  const idCol       = headers.indexOf('ID')
  const colorCol    = headers.indexOf('Couleur')
  const variantCol  = headers.indexOf('VariantId')

  const debug = {
    idRecherche: data.id,
    couleurRecherchee: data.Couleur || data.color || '',
    variantIdRecherche: data.variantId || data.VariantId || '',
    lignesDansLaFeuille: allData.length - 1,
  }

  // 1) Recherche par VariantId si fourni
  const wantedVariantId = data.variantId || data.VariantId
  if (wantedVariantId && variantCol !== -1) {
    for (let i = 1; i < allData.length; i++) {
      if (String(allData[i][variantCol]).trim() === String(wantedVariantId).trim()) {
        Logger.log('[findVariantRow] trouvé par VariantId=' + wantedVariantId + ' → ligne ' + (i + 1))
        return { rowIndex: i + 1, headers, allData, debug, matchedBy: 'variantId' }
      }
    }
    Logger.log('[findVariantRow] VariantId=' + wantedVariantId + ' non trouvé, fallback sur ID+Couleur')
  }

  // 2) Fallback : ID + Couleur normalisés (trim + lowercase)
  const wantedColor = data.Couleur != null ? data.Couleur : data.color
  for (let i = 1; i < allData.length; i++) {
    if (norm(allData[i][idCol]) === norm(data.id) &&
        norm(allData[i][colorCol]) === norm(wantedColor)) {
      Logger.log('[findVariantRow] trouvé par ID+Couleur normalisés → ligne ' + (i + 1))
      return { rowIndex: i + 1, headers, allData, debug, matchedBy: 'id+couleur' }
    }
  }

  Logger.log('[findVariantRow] AUCUNE variante trouvée. debug=' + JSON.stringify(debug))
  return { rowIndex: -1, headers, allData, debug, matchedBy: null }
}

// ── Modifier un produit ───────────────────────────────────────
function updateProduct(data) {
  const sheet = ensureStockSheet()
  Logger.log('[updateProduct] données reçues = ' + JSON.stringify(data))

  const found = findVariantRow(sheet, data)
  if (found.rowIndex === -1) {
    return json({
      success: false,
      error: 'Variante introuvable (ID + Couleur).',
      debug: found.debug,
    })
  }

  const targetRow = found.rowIndex
  const headers   = found.headers

  headers.forEach((h, j) => {
    if (h === 'ID') return
    if (h === 'VariantId') {
      // Ne jamais écraser un VariantId existant ; le générer s'il manquait
      const current = found.allData[targetRow - 1][j]
      if (!current) sheet.getRange(targetRow, j + 1).setValue(Utilities.getUuid())
      return
    }
    if (h === 'Mise à jour') {
      sheet.getRange(targetRow, j + 1).setValue(
        Utilities.formatDate(new Date(), 'Africa/Casablanca', 'yyyy-MM-dd')
      )
    } else if (h === 'Prix' || h === 'PrixPromo') {
      if (data[h] != null && data[h] !== '') {
        sheet.getRange(targetRow, j + 1).setValue(Number(data[h]))
      }
    } else if (ALL_SIZES.includes(h)) {
      if (data[h] != null) sheet.getRange(targetRow, j + 1).setValue(Number(data[h]))
    } else if (data[h] != null) {
      sheet.getRange(targetRow, j + 1).setValue(data[h])
    }
  })

  Logger.log('[updateProduct] ligne ' + targetRow + ' mise à jour (trouvée par ' + found.matchedBy + ')')
  clearProductsCache()
  return json({ success: true, updated: true })
}

// ── Téléverser une image (Drive) ───────────────────────────────
// data = { action:'uploadImage', filename, mimeType, base64 }
// Retourne { success, url, fileId } — url = lien direct utilisable en <img>
function uploadImage(data) {
  if (!data.base64) return json({ success: false, error: 'Aucune image reçue.' })

  try {
    const folder = getOrCreateImagesFolder()
    const mime   = data.mimeType || 'image/jpeg'
    const bytes  = Utilities.base64Decode(data.base64.replace(/^data:[^,]+,/, ''))
    const blob   = Utilities.newBlob(bytes, mime, data.filename || ('image-' + Date.now() + '.jpg'))
    const file   = folder.createFile(blob)
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW)

    const fileId = file.getId()
    const url    = 'https://lh3.googleusercontent.com/d/' + fileId + '=w1600'
    return json({ success: true, url: url, fileId: fileId })
  } catch (err) {
    return json({ success: false, error: err.toString() })
  }
}

function getOrCreateImagesFolder() {
  const folders = DriveApp.getFoldersByName(IMAGES_FOLDER)
  if (folders.hasNext()) return folders.next()
  return DriveApp.createFolder(IMAGES_FOLDER)
}

// ── Supprimer un produit ──────────────────────────────────────
// Sans ?color → supprime TOUTES les variantes de cet ID
// Avec ?color → supprime uniquement cette couleur
// Avec data.soft=true → suppression logique (Statut='supprime') au lieu
// d'une suppression physique de la ligne (utile pour garder l'historique)
function deleteProduct(data) {
  const sheet   = ensureStockSheet()
  const allData = sheet.getDataRange().getValues()
  const headers = allData[0]
  const idCol      = headers.indexOf('ID')
  const colorCol   = headers.indexOf('Couleur')
  const statutCol  = headers.indexOf('Statut')
  const variantCol = headers.indexOf('VariantId')

  Logger.log('[deleteProduct] données reçues = ' + JSON.stringify(data))

  const wantedVariantId = data.variantId || data.VariantId
  const toDelete = []
  for (let i = allData.length - 1; i >= 1; i--) {
    let isMatch
    if (wantedVariantId) {
      isMatch = String(allData[i][variantCol]).trim() === String(wantedVariantId).trim()
    } else {
      const matchId    = norm(allData[i][idCol]) === norm(data.id)
      const matchColor = !data.color || norm(allData[i][colorCol]) === norm(data.color)
      isMatch = matchId && matchColor
    }
    if (isMatch) toDelete.push(i + 1)
  }

  if (toDelete.length === 0) {
    Logger.log('[deleteProduct] AUCUNE variante trouvée pour id=' + data.id + ' color=' + data.color)
    return json({
      success: false,
      error: 'Variante introuvable.',
      debug: { idRecherche: data.id, couleurRecherchee: data.color || '', variantIdRecherche: wantedVariantId || '' },
    })
  }

  if (data.soft) {
    toDelete.forEach(r => sheet.getRange(r, statutCol + 1).setValue('supprime'))
    Logger.log('[deleteProduct] suppression logique (Statut=supprime) sur ' + toDelete.length + ' ligne(s)')
    clearProductsCache()
    return json({ success: true, deleted: toDelete.length, soft: true })
  }

  toDelete.sort((a, b) => b - a).forEach(r => sheet.deleteRow(r))
  Logger.log('[deleteProduct] ' + toDelete.length + ' ligne(s) supprimée(s) définitivement')
  clearProductsCache()
  return json({ success: true, deleted: toDelete.length })
}

function updateStock(data) {
  const sheet   = ensureStockSheet()
  const allData = sheet.getDataRange().getValues()
  const headers = allData[0]
  const idCol    = headers.indexOf('ID')
  const colorCol = headers.indexOf('Couleur')
  const majCol   = headers.indexOf('Mise à jour')

  let targetRow = -1
  for (let i = 1; i < allData.length; i++) {
    if (String(allData[i][idCol])    === String(data.id) &&
        String(allData[i][colorCol]) === String(data.color)) {
      targetRow = i + 1
      break
    }
  }

  if (targetRow === -1) {
    const newRow = new Array(headers.length).fill('')
    newRow[idCol]                         = data.id
    newRow[headers.indexOf('Produit')]    = data.name       || ''
    newRow[headers.indexOf('Collection')] = data.collection || ''
    newRow[colorCol]                      = data.color      || ''
    if (majCol !== -1) newRow[majCol] = Utilities.formatDate(new Date(), 'Africa/Casablanca', 'yyyy-MM-dd')
    ALL_SIZES.forEach(size => {
      const ci = headers.indexOf(size)
      if (ci !== -1) newRow[ci] = (data.sizeStocks && data.sizeStocks[size] != null) ? Number(data.sizeStocks[size]) : ''
    })
    sheet.appendRow(newRow)
    clearProductsCache()
    return json({ success: true, created: true })
  }

  ALL_SIZES.forEach(size => {
    const ci = headers.indexOf(size)
    if (ci !== -1 && data.sizeStocks && data.sizeStocks[size] != null)
      sheet.getRange(targetRow, ci + 1).setValue(Number(data.sizeStocks[size]))
  })
  if (majCol !== -1) sheet.getRange(targetRow, majCol + 1).setValue(
    Utilities.formatDate(new Date(), 'Africa/Casablanca', 'yyyy-MM-dd')
  )
  clearProductsCache()
  return json({ success: true, updated: true })
}

//  COMMANDES — enregistrer + décrémenter stock
// ════════════════════════════════════════════════════════════════

function saveOrder(data) {
  const ss    = SpreadsheetApp.getActiveSpreadsheet()
  const sheet = ss.getSheetByName(ORDERS_SHEET)
  const produits = data.produits || []
  if (produits.length === 0) return json({ result: 'error', message: 'Aucun produit dans la commande' })

  produits.forEach(p => {
    sheet.appendRow([
      new Date(),
      data.nom        || '',
      data.telephone  || '',
      data.ville      || '',
      data.adresse    || '',
      p.modele        || '',
      p.taille        || '',
      p.couleur       || '',
      p.quantite      || 1,
      p.prixUnitaire  || 0,
      data.sousTotal  || data.prix || 0,
      data.livraison  || 0,
      data.prix       || 0,
      '',
      data.paiement   || 'Paiement à la livraison',
      'En attente',
    ])
  })

  produits.forEach(p => decrementStock(p.modele, p.couleur, p.taille, p.quantite || 1))
  return json({ result: 'success' })
}

function decrementStock(modele, couleur, taille, quantite) {
  try {
    const ss         = SpreadsheetApp.getActiveSpreadsheet()
    const sheet      = ss.getSheetByName(STOCK_SHEET)
    if (!sheet) return
    const allData    = sheet.getDataRange().getValues()
    const headers    = allData[0]
    const produitCol = headers.indexOf('Produit')
    const colorCol   = headers.indexOf('Couleur')
    const sizeCol    = headers.indexOf(taille)
    if (sizeCol === -1) return
    for (let i = 1; i < allData.length; i++) {
      if (String(allData[i][produitCol]).toLowerCase() === String(modele).toLowerCase() &&
          String(allData[i][colorCol]).toLowerCase()   === String(couleur).toLowerCase()) {
        const current = Number(allData[i][sizeCol]) || 0
        sheet.getRange(i + 1, sizeCol + 1).setValue(Math.max(0, current - quantite))
        break
      }
    }
  } catch (e) { /* silencieux */ }
}
